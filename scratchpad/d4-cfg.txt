   170	async def _performs(
   171	    produced: Reply | Awaitable[Reply],
   172	    *,
   173	    ask: Callable[[Question], Awaitable[Answer | None]],
   174	    call: Callable[[str, Mapping[str, JsonValue]], Awaitable[ToolResult]],
   175	    report: Callable[[str], None],
   176	) -> AgentOutcome:
   177	    reply = await produced if isawaitable(produced) else produced
   178	    for line in reply.activity:
   179	        report(line)
   180	    for question in reply.asks:
   181	        await ask(question)
   182	    for made in reply.calls:
   183	        await call(made.tool, made.payload)
   184	    return AgentOutcome(stop_reason=reply.stop_reason, text=reply.says)

     8	__all__ = ["GROUP", "installed", "load", "names"]

    10	GROUP: Final = "agl.workflows"
    11	
    12	
    13	def installed() -> tuple[EntryPoint, ...]:
    14	    return tuple(entry_points(group=GROUP))
    15	
    16	
    17	def names(points: Iterable[EntryPoint]) -> tuple[str, ...]:
    18	    return tuple(sorted(_index(points)))
    19	
    20	
    21	def load[T](points: Iterable[EntryPoint], name: str, kind: type[T]) -> T:
    22	    index = _index(points)
    23	    point = index.get(name)
    24	    if point is None:
    25	        raise NotFoundError(_unknown(name, index))
    26	    try:
    27	        loaded = point.load()
    28	    except (ImportError, AttributeError) as error:
    29	        raise InputError(
    30	            f"the workflow {name!r} is registered as {point.value!r}, and loading it failed: "
    31	            f"{error}. That declaration is in the {GROUP} entry points of the package that "
    32	            f"provides {name!r}, so the fix is in that package or in how it is installed - AGL "
    33	            f"only read what it declared. The original error is chained below this one"
    34	        ) from error
    35	    if not isinstance(loaded, kind):
    36	        raise InputError(
    37	            f"the workflow {name!r} is registered as {point.value!r}, which loaded and turned "
    38	            f"out to be a {_describe(type(loaded))} rather than a {_describe(kind)}. The "
    39	            f"{GROUP} entry point of the package that provides {name!r} is pointing at the wrong "
    40	            f"object - AGL only read what it declared"
    41	        )
    42	    return loaded
    43	
    44	
    45	def _index(points: Iterable[EntryPoint]) -> Mapping[str, EntryPoint]:
    46	    index: dict[str, EntryPoint] = {}
    47	    for point in points:
    48	        held = index.get(point.name)
    49	        if held is not None:
    50	            raise ConflictError(
    51	                f"two installed packages both register a workflow named {point.name!r}, as "
    52	                f"{held.value!r} and as {point.value!r}. AGL will not choose between them: running "
    53	                f"the wrong one of two workflows that answer to the same name is a mistake nothing "
    54	                f"downstream could report, so the name stays ambiguous until one of the two "
    55	                f"packages is uninstalled"
    56	            )
    57	        index[point.name] = point
    58	    return index
    59	
    60	
    61	def _unknown(name: str, index: Mapping[str, EntryPoint]) -> str:
    62	    registered = tuple(sorted(index))
    63	    if not registered:
    64	        return (
    65	            f"there is no workflow named {name!r}, and in fact no workflow is installed at all: "
    66	            f"the {GROUP} entry point group is empty in this environment. A workflow arrives as a "
    67	            f"package that declares one entry point in that group; install one, then `agl "
    68	            f"workflows` lists it"
    69	        )
    70	    return (
    71	        f"there is no workflow named {name!r}. Installed and registered under {GROUP}: "
    72	        f"{', '.join(registered)}. `agl workflows` prints the same list"
    73	    )
    74	
    75	
    76	def _describe(kind: type[object]) -> str:
    77	    return f"{kind.__module__}.{kind.__qualname__}"

    81	                f"the build is allowed to take, so it must be finite and greater than zero. Zero "
    82	                f"or less kills every build before it starts; an infinite deadline is a run that "
    83	                f"hangs on a build nobody is watching"

    91	            f"relative program path resolves against the process's working directory - which for "
    92	            f"an agent run is a worktree AGL chose. Give an absolute path, or leave it unset and "
    93	            f"let the adapter resolve the harness from PATH"
    94	        )

    75	def resolve_settings(overrides: Overrides, environ: Mapping[str, str]) -> Settings:
    76	    flagged = _first(overrides.home, _path(environ, _variable(_HOME)))
    77	    home = AglHome(_default_home(environ) if flagged is None else flagged)
    78	    said = toml_file.read_settings(home)
    79	    claude = _agent(
    80	        _CLAUDE, overrides.claude_enabled, overrides.claude_cli_path, said.claude, environ
    81	    )
    82	    openai = _agent(
    83	        _OPENAI, overrides.openai_enabled, overrides.openai_cli_path, said.openai, environ
    84	    )
    85	    return Settings(
    86	        home=home,
    87	        agents=AgentSettings(
    88	            claude=ClaudeSettings(enabled=claude[0], cli_path=claude[1]),
    89	            openai=OpenAiSettings(enabled=openai[0], cli_path=openai[1]),
    90	        ),
    91	    )

    94	def resolve_project(
    95	    settings: Settings, overrides: Overrides, environ: Mapping[str, str], cwd: Path

    97	    said = toml_file.resolve_project(settings.home, cwd)
    98	    path = project_config(settings.home, said.name)
    99	    return Project(
   100	        name=said.name,
   101	        repo=_required(said.repo, path, _REPO),
   102	        trees=_required(said.trees_root, path, _TREES_ROOT),
   103	        build=_required(
   104	            _first(overrides.build, _text(environ, _variable(_BUILD)), said.build), path, _BUILD
   105	        ),
   106	        build_timeout=_settled(
   107	            overrides.build_timeout,
   108	            _seconds(environ, _variable(_BUILD_TIMEOUT)),
   109	            said.build_timeout,
   110	            default=DEFAULT_BUILD_TIMEOUT,
   111	        ),
   112	    )

   115	def _first[T](*layers: T | None) -> T | None:
   116	    for value in layers:
   117	        if value is not None:
   118	            return value
   119	    return None
   120	
   121	
   122	def _settled[T](*layers: T | None, default: T) -> T:
   123	    answer = _first(*layers)
   124	    return default if answer is None else answer

   126	
   127	def _required[T](answer: T | None, path: Path, key: str) -> T:
   128	    if answer is None:
   129	        raise InputError(
   130	            f"{path}: {key} is not set, and there is no default AGL could apply - it is a fact "
   131	            f"about this project that only this file holds. `agl init` inside the repository "
   132	            f"writes the file with all five of its keys; add {key} to it, or run init again"
   133	        )
   134	    return answer

   137	def _agent(
   138	    section: str,
   139	    enabled: bool | None,
   140	    cli_path: Path | None,
   141	    said: toml_file.FileAgent,
   142	    environ: Mapping[str, str],
   143	) -> tuple[bool, Path | None]:
   144	    return (
   145	        _settled(
   146	            enabled,
   147	            _flag(environ, _variable(_AGENT, section, _ENABLED)),
   148	            said.enabled,
   149	            default=_DEFAULT_ENABLED,
   150	        ),
   151	        _first(cli_path, _path(environ, _variable(_AGENT, section, _CLI_PATH)), said.cli_path),
   152	    )

   155	def _variable(*path: str) -> str:
   156	    return _PREFIX + "_".join(path).upper()

   158	
   159	def _default_home(environ: Mapping[str, str]) -> Path:
   160	    said = environ.get(_USER_HOME, "").strip()
   161	    if said and Path(said).is_absolute():
   162	        return Path(said) / _DEFAULT_HOME_DIRNAME
   163	    raise InputError(
   164	        f"AGL cannot tell where to keep its own state: {_variable(_HOME)} is unset, and the "
   165	        f"default {_DEFAULT_HOME_DIRNAME} directory under your home cannot be located because "
   166	        f"{_USER_HOME} is {said!r} rather than an absolute path. Set {_variable(_HOME)} to an "
   167	        f"absolute directory"
   168	    )

   170	
   171	def _text(environ: Mapping[str, str], variable: str) -> str | None:
   172	    if variable not in environ:
   173	        return None
   174	    value = environ[variable].strip()
   175	    if not value:
   176	        raise InputError(
   177	            f"{variable} is set to the empty string, which is not a value AGL can use. An empty "
   178	            f"variable reads as an operator clearing a setting, so it is refused rather than "
   179	            f"treated as unset - to let the settings file or the default decide, unset it"
   180	        )
   181	    return value

   184	def _flag(environ: Mapping[str, str], variable: str) -> bool | None:
   185	    value = _text(environ, variable)
   186	    if value is None:
   187	        return None
   188	    if value.lower() in (_TRUE, _FALSE):
   189	        return value.lower() == _TRUE
   190	    raise _wrong(variable, value, f"{_TRUE} or {_FALSE}")

   192	
   193	def _seconds(environ: Mapping[str, str], variable: str) -> float | None:
   194	    value = _text(environ, variable)
   195	    if value is None:
   196	        return None
   197	    try:
   198	        return float(value)
   199	    except ValueError as error:
   200	        raise _wrong(variable, value, "a number of seconds") from error
   201	
   202	
   203	def _path(environ: Mapping[str, str], variable: str) -> Path | None:
   204	    value = _text(environ, variable)
   205	    if value is None:
   206	        return None
   207	    path = Path(value)
   208	    if not path.is_absolute():
   209	        raise InputError(
   210	            f"{variable} is {value!r}, which is a relative path. Every path AGL is given is "
   211	            f"absolute, because a relative one resolves against whatever directory AGL was invoked "
   212	            f"from - for an agent step, a worktree AGL chose. A leading ~ counts: the shell "
   213	            f"expands it, and a variable set without one arrives here unexpanded"
   214	        )
   215	    return path
   216	
   217	
   218	def _wrong(variable: str, value: str, expected: str) -> InputError:
   219	    return InputError(
   220	        f"{variable} is {value!r}, which is not {expected}. AGL refuses it rather than falling "
   221	        f"through to the settings file or the default, because a typo that looks like a setting "
   222	        f"nobody set is the failure configuration exists to prevent"
   223	    )

    56	        self, role: Role[R], *, commit: str | None, inputs: Mapping[str, object]
    57	    ) -> R:
    58	        step = StepName(role.name)
    59	        await self._capabilities.require(self._services.agents, role, step=str(step))
    60	        journal, workspace = await self._namespace()
    61	
    62	        capture: _Capture[R] | None = None
    63	        tools: list[Tool] = []
    64	        for declared in role.tools:
    65	            if isinstance(declared, ReportingTool):
    66	                capture = _Capture(declared)
    67	                tools.append(capture.tool)
    68	            else:
    69	                tools.append(declared)
    70	        offered = tuple(tools)

    72	        async def _worker() -> JsonValue:
    73	            try:
    74	                outcome = await self._services.agents.run(
    75	                    AgentTask(
    76	                        instructions=_composed(role.instructions, inputs),
    77	                        workspace=workspace.path,
    78	                        model=role.model,
    79	                        restrictions=frozenset(role.restrictions),
    80	                        tools=offered,
    81	                    ),
    82	                    on_question=role.on_question,
    83	                    on_activity=self._reported,
    84	                )
    85	            finally:
    86	                self._activity = None
    87	            return None if capture is None else capture.reported(outcome)
    88	
    89	        value = await journal.step(
    90	            step,
    91	            instructions=role.instructions,
    92	            model=role.model,
    93	            restrictions=role.restrictions,
    94	            tools=offered,
    95	            inputs=inputs,
    96	            worker=_worker,
    97	            commit=commit,
    98	        )

   100	            return cast(R, None)
   101	        return capture.read(value)

   105	            return self._opened

   108	                base = await self._services.history.resolve(self._base)
   109	                workspace = await self._services.workspaces.open(
   110	                    self._scope.label, _namespace_of(self._scope), base
   111	                )
   112	                self._opened = (
   113	                    Journal(
   114	                        self._services.store,
   115	                        self._scope,
   116	                        workspace,
   117	                        self._services.clock,
   118	                        self._fingerprints,
   119	                        base,
   120	                    ),
   121	                    workspace,
   122	                )
   123	            return self._opened

   126	class _Capture[P]:
   127	
   128	    def __init__(self, declaration: ReportingTool[P]) -> None:
   129	        self._declaration = declaration
   130	        self._payload: dict[str, JsonValue] | None = None
   131	        self.tool = Tool(
   132	            name=declaration.name,
   133	            description=declaration.description,
   134	            payload_schema=declaration.payload_schema,
   135	            handler=self._called,
   136	        )
   137	
   138	    async def _called(self, payload: Mapping[str, JsonValue]) -> ToolResult:
   139	        if self._payload is not None:
   140	            return ToolResult(
   141	                text=(
   142	                    f"{self._declaration.name} has already recorded this step's result: it records "
   143	                    f"one payload per run, and the first one stands. This call changed nothing."
   144	                ),
   145	                rejected=True,
   146	            )
   147	        rejection = self._declaration.rejection(payload)
   148	        if rejection is not None:
   149	            return ToolResult(text=rejection, rejected=True)
   150	        self._payload = dict(payload)
   151	        return ToolResult(
   152	            text=(
   153	                f"{self._declaration.name} recorded this step's result. Nothing further is needed "
   154	                f"from this tool."
   155	            )
   156	        )
   157	
   158	    def reported(self, outcome: AgentOutcome) -> JsonValue:
   159	        if self._payload is None:
   160	            raise RoleIncompleteError(_unreported(self._declaration.name, outcome))
   161	        return self._payload
   162	
   163	    def read(self, value: object) -> P:
   164	        return self._declaration.read(value)

   168	    if not inputs:
   169	        return instructions
   170	    return "\n\n".join((instructions, _INPUTS_HEADING, canonical_json(inputs)))

   178	    return (
   179	        f"the agent finished without ever calling {tool!r}, so this step produced no result: "
   180	        f"nothing was recorded and it will run again on the next attempt. {_because(outcome)} It "
   181	        f"said this instead of reporting: {outcome.text!r}"

   185	def _because(outcome: AgentOutcome) -> str:
   186	    if outcome.stop_reason is StopReason.LIMIT:
   187	        return (
   188	            "The backend stopped it against its will - turns, tokens, time or budget - so it may "
   189	            "simply have run out of room before it reported: raise the limit."
   190	        )
   191	    if outcome.stop_reason is StopReason.COMPLETED:
   192	        return (
   193	            "It ended its own turn, having decided it was finished, so the limit is not what it "
   194	            "reached: the prompt is what did not read as asking for a report through that tool."
   195	        )
   196	    return (
   197	        "The backend did not say why it stopped, so neither a limit it reached nor a prompt that "
   198	        "never asked for the report can be ruled out from here."
   199	    )

    15	_FROM_ON_QUESTION: Final = (
    16	    f". {str(Capability.MID_RUN_QUESTIONS)!r} is in this role's `requires` because it declares "
    17	    f"`on_question`: `sdk/roles.py` folds it in at declaration time, so there is no line in the "
    18	    f"workflow to go looking for. A backend that cannot ask would not block on the question - it "
    19	    f"would tell the agent no answer is available, leaving the approval gate silently absent and "
    20	    f"the step reporting a result anyway (§3.7)"
    21	)
    22	
    23	_FROM_TOOLS: Final = (
    24	    f". {str(Capability.TOOL_CALLING)!r} is in this role's `requires` because it declares `tools`: "
    25	    f"`sdk/roles.py` folds it in at declaration time, so there is no line in the workflow to go "
    26	    f"looking for. Either this role names a model whose backend can call a tool, or it offers none "
    27	    f"- and a role with no tools is an effect step, whose result is `null` (§3.3)"
    28	)

    31	class Capabilities:
    32	
    33	    def __init__(self) -> None:
    34	        self._known: dict[ModelId, frozenset[Capability]] = {}
    35	
    36	    async def require(self, runner: AgentRunner, role: Role[object], *, step: str) -> None:
    37	        held = self._known.get(role.model)
    38	        if held is None:
    39	            held = await runner.capabilities(role.model)
    40	            self._known[role.model] = held
    41	        missing = frozenset(role.requires) - held
    42	        if missing:
    43	            raise DeniedError(_unmet(step, role, missing, held))

    45	
    46	async def check(runner: AgentRunner, declared_by: _Declaration) -> None:
    47	    for factory in _demanded(_declared_beside(declared_by)):
    48	        try:
    49	            await runner.check_ready(factory.model)
    50	        except UpstreamUnavailable as unavailable:
    51	            raise UpstreamUnavailable(
    52	                _not_ready(unavailable, factory, declared_by.__module__)
    53	            ) from unavailable
    54	
    55	
    56	def _declared_beside(declared_by: _Declaration) -> tuple[RoleFactory[..., Any], ...]:
    57	    written_in = sys.modules.get(declared_by.__module__)
    58	    if written_in is None:
    59	        raise InternalError(
    60	            f"the workflow function {declared_by.__qualname__!r} says it was written in "
    61	            f"{declared_by.__module__!r}, and that module is not in `sys.modules`. "
    62	            f"Preflight reads a workflow's role factories out of the namespace its `def` ran in "
    63	            f"(§3.11), and an entry point is what imported that module, so there is no supported "
    64	            f"way to reach this line"
    65	        )
    66	    beside = tuple(vars(written_in).values())
    67	    return tuple(bound for bound in beside if isinstance(bound, RoleFactory)) + tuple(
    68	        inside
    69	        for bound in beside
    70	        if isinstance(bound, ModuleType)
    71	        for inside in vars(bound).values()
    72	        if isinstance(inside, RoleFactory)
    73	    )
    74	
    75	
    76	def _demanded(factories: tuple[RoleFactory[..., Any], ...]) -> tuple[RoleFactory[..., Any], ...]:
    77	    first: dict[ModelId, RoleFactory[..., Any]] = {}
    78	    for factory in factories:
    79	        first.setdefault(factory.model, factory)
    80	    return tuple(first.values())

    82	
    83	def _not_ready(
    84	    refusal: UpstreamUnavailable, factory: RoleFactory[..., Any], workflow_module: str
    85	) -> str:
    86	    return (
    87	        f"{refusal} - and AGL asked because {str(factory.model)!r} is the model of the role "
    88	        f"factory `{factory.name}`, declared in {factory.__module__!r} and reached from "
    89	        f"{workflow_module!r}, which is the module this run's workflow is written in. Preflight "
    90	        f"reads the `@role(model=…)` factories in that namespace - and in any module bound in it - "
    91	        f"and asks each distinct model's backend whether it is ready, without calling any of them "
    92	        f"(§3.2). That scan over-approximates, deliberately: a role imported into a workflow's "
    93	        f"module and never stepped with still demands its provider here, and so does every other "
    94	        f"role of a module imported for one of them, which is the known cost of one declaration "
    95	        f"instead of two. If nothing in {workflow_module!r} steps with `{factory.name}`, this is a "
    96	        f"false refusal and the line that put the demand there is an import - of `{factory.name}` "
    97	        f"itself, or of a module that binds it - so dropping that import drops this demand; "
    98	        f"otherwise the harness above is the thing to fix"
    99	    )

   102	def _unmet(
   103	    step: str, role: Role[object], missing: frozenset[Capability], held: frozenset[Capability]
   104	) -> str:
   105	    wanted = sorted(str(member) for member in missing)
   106	    offered = sorted(str(member) for member in held)
   107	    message = (
   108	        f"the role handed to step {step!r} cannot run on {str(role.model)!r}: it requires "
   109	        f"{wanted}, which the backend serving that model does not offer - it reports {offered}. "
   110	        f"A capability "
   111	        f"is what a backend can be asked for at all (§3.2), so this does not clear up on its own: "
   112	        f"either the role names a model whose backend has it, or it stops requiring it"
   113	    )
   114	    if Capability.MID_RUN_QUESTIONS in missing and role.on_question is not None:
   115	        message += _FROM_ON_QUESTION
   116	    if Capability.TOOL_CALLING in missing and role.tools:
   117	        message += _FROM_TOOLS
   118	    return message

    12	class Worktrees[R]:
    13	
    14	    def __init__(self) -> None:
    15	        self._taken: dict[str, _Taken[R]] = {}
    16	
    17	    def open(
    18	        self, name: str, *, scope: RunScope, base: str, build: Callable[[RunScope, str], R]
    19	    ) -> R:
    20	        namespace = Namespace(name)
    21	        key = namespace.collision_key
    22	        taken = self._taken.get(key)
    23	        if taken is not None:
    24	            if taken.scope == scope and taken.namespace == namespace:
    25	                return taken.child
    26	            raise ConflictError(_collision(namespace, scope, taken.namespace, taken.scope))
    27	        child = build(scope.inside(namespace), base)
    28	        self._taken[key] = _Taken(namespace, scope, child)
    29	        return child
    30

    32	@dataclass(frozen=True, slots=True)
    33	class _Taken[R]:
    34	
    35	    namespace: Namespace
    36	
    37	    scope: RunScope
    38	
    39	    child: R
    40	
    41	
    42	def _collision(namespace: Namespace, scope: RunScope, held: Namespace, holder: RunScope) -> str:
    43	    return (
    44	        f"namespace {str(namespace)!r}, asked for by {_where(scope)}, is already taken in run "
    45	        f"{str(holder.label)!r}: {_where(holder)} holds it as {str(held)!r}. Namespace names are "
    46	        f"unique run-wide and not merely among siblings (§3.9), compared case-insensitively, "
    47	        f"because the trees root is flat - both of these are one checkout directory in this run, "
    48	        f"and whichever opened second would be handed the other's working tree. Nothing was "
    49	        f"changed: pick another name"
    50	    )
    51	
    52	
    53	def _where(scope: RunScope) -> str:
    54	    if not scope.namespaces:
    55	        return "the run itself"
    56	    return "the worktree " + " -> ".join(str(name) for name in scope.namespaces)
