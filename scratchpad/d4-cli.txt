    26	__all__ = ["Compose", "Invocation", "main", "parser"]
    27	
    28	_PROGRAM: Final = "agl"
    29	
    30	_COMMAND: Final = "command"
    31	
    32	_DESCRIPTION: Final = "Run AI agent workflows against a code repository."
    33	
    34	_OUR_BUG: Final = (
    35	    "that traceback is AGL's own bug and not something you did wrong. An adapter is meant to "
    36	    "translate what it catches into an `agl.ports.errors` class at its boundary, and one did not - "
    37	    "so nothing above it could report the failure in words. Please report it with the lines above"
    38	)

    41	@dataclass(frozen=True, slots=True)
    42	class Invocation:
    43	
    44	    registered: commands.Registered
    45	
    46	    settings: Settings
    47	
    48	    cwd: Path
    49	
    50	    points: Iterable[EntryPoint] | None = None
    51	
    52	    ask: Ask = input
    53	
    54	
    55	type Compose = Callable[[], Invocation]
    56	
    57	
    58	def main(argv: Sequence[str] | None = None, *, compose: Compose | None = None) -> int:
    59	    try:
    60	        parsed, tail = parser().parse_known_args(argv)
    61	        invocation = _compose() if compose is None else compose()
    62	        return _dispatch(invocation, parsed, tail)
    63	    except Stop as stop:
    64	        print(f"stopped: {stop}")
    65	        return exit_status(stop)
    66	    except AglError as refusal:
    67	        print(f"{_PROGRAM}: {refusal}", file=sys.stderr)
    68	        return exit_status(refusal)
    69	    except ExceptionGroup as concurrent:
    70	        return _concurrent(concurrent)
    71	    except Exception as bug:
    72	        print_exception(bug, file=sys.stderr)
    73	        print(f"{_PROGRAM}: {_OUR_BUG}", file=sys.stderr)
    74	        return exit_status(bug)
    75	
    76	
    77	def parser() -> RefusingParser:
    78	    root = RefusingParser(prog=_PROGRAM, description=_DESCRIPTION, allow_abbrev=False)
    79	    declared = root.add_subparsers(
    80	        dest=_COMMAND, metavar="<command>", required=True, parser_class=RefusingParser
    81	    )
    82	    run_command.declare(declared)
    83	    resume_command.declare(declared)
    84	    clear_command.declare(declared)
    85	    init_command.declare(declared)
    86	    workflows_command.declare(declared)
    87	    return root
    88	
    89	
    90	def _compose() -> Invocation:
    91	    resolved = sources.resolve(sources.Overrides())
    92	    cwd = Path.cwd()
    93	    return Invocation(
    94	        registered=lambda: _registered(resolved, cwd), settings=resolved.settings, cwd=cwd
    95	    )
    96	
    97	
    98	def _registered(resolved: sources.Resolved, cwd: Path) -> tuple[ProjectName, Services]:
    99	    project = resolved.project(cwd)
   100	    return project.name, container.real(resolved.settings, project)
   101	
   102	
   103	def _dispatch(invocation: Invocation, parsed: argparse.Namespace, tail: Sequence[str]) -> int:
   104	    command = getattr(parsed, _COMMAND)
   105	    if command == run_command.NAME:
   106	        return run_command.execute(invocation.registered, parsed, tail, points=invocation.points)
   107	    if command == resume_command.NAME:
   108	        _no_tail(command, tail)
   109	        return resume_command.execute(invocation.registered, parsed, points=invocation.points)
   110	    if command == clear_command.NAME:
   111	        _no_tail(command, tail)
   112	        return clear_command.execute(invocation.registered, parsed)
   113	    if command == init_command.NAME:
   114	        _no_tail(command, tail)
   115	        return init_command.execute(invocation.settings, invocation.cwd, invocation.ask)
   116	    if command == workflows_command.NAME:
   117	        _no_tail(command, tail)
   118	        return workflows_command.execute(parsed, points=invocation.points)
   119	    raise InternalError(
   120	        f"`{_PROGRAM} {command}` reached the dispatch and there is no command by that name. The "
   121	        f"parser admits only the subcommands declared in this module, so this is AGL's own bug "
   122	        f"rather than anything you typed - a command was declared and never given a clause here"
   123	    )
   124	
   125	
   126	def _no_tail(command: str, tail: Sequence[str]) -> None:
   127	    if tail:
   128	        raise InputError(
   129	            f"`{_PROGRAM} {command}` does not take {list(tail)}: the arguments on this line are "
   130	            f"the ones `{_PROGRAM} {command} -h` lists and no others. `{_PROGRAM} run` is the one "
   131	            f"command that carries arguments AGL does not understand, because it is the one that "
   132	            f"names a workflow to hand them to - a run's own workflow, its base ref and its "
   133	            f"parameters are read back afterwards from the record `{_PROGRAM} run` wrote, and the "
   134	            f"rest of AGL's commands either address a run that already has one or address no run "
   135	            f"at all. To run a workflow with parameters, start a run: `{_PROGRAM} run <workflow> "
   136	            f"-n <label> [workflow flags]`; `{_PROGRAM} workflows <workflow>` lists the flags one "
   137	            f"takes"
   138	        )
   139	
   140	
   141	def _concurrent(group: ExceptionGroup[Exception]) -> int:
   142	    deliberate, failed = group.split(Stop)
   143	    if deliberate is not None:
   144	        for stop in leaves(deliberate):
   145	            print(f"stopped: {stop}")
   146	    if failed is not None:
   147	        _, untranslated = failed.split(AglError)
   148	        if untranslated is not None:
   149	            print_exception(untranslated, file=sys.stderr)
   150	            print(f"{_PROGRAM}: {_OUR_BUG}", file=sys.stderr)
   151	        print(f"{_PROGRAM}: {_severally(group)}", file=sys.stderr)
   152	    return exit_status(group)
   153	
   154	
   155	def _severally(group: ExceptionGroup[Exception]) -> str:
   156	    held = tuple(leaves(group))
   157	    status = exit_status(group)
   158	    named = "\n".join(f"    {exit_status(one)}  {type(one).__name__}: {one}" for one in held)
   159	    if {exit_status(one) for one in held} == {status}:
   160	        return (
   161	            f"this run's concurrent children raised, and every one of them below is exit {status} "
   162	            f"- so that is what the run exits with, exactly as any one of them raised on its own "
   163	            f"would have:\n{named}"
   164	        )
   165	    return (
   166	        f"this run's concurrent children raised, and they do not resolve to one exit status - so "
   167	        f"the run exits {status}. A run that failed several different ways is not attributable to "
   168	        f"one code (§3.1), and `InternalError`'s is the honest answer rather than a guess at which "
   169	        f"of them was the real one, so read this {status} as 'these disagreed' and not as its "
   170	        f"usual 'file a bug'. All of them, with the status each resolves to on its own:\n{named}"
   171	    )

    19	_LABEL: Final = "label"
    20	_BASE_REF: Final = "base_ref"

    48	        *_LABEL_FLAGS,
    49	        dest=_LABEL,
    50	        metavar="<label>",

    52	        help="this run's name: its record under AGL_HOME and the branch agl/<label>",

    56	        dest=_BASE_REF,

    63	def execute(
    64	    registered: Registered,
    65	    parsed: argparse.Namespace,
    66	    argv: Sequence[str],
    67	    *,
    68	    points: Iterable[EntryPoint] | None = None,
    69	) -> int:
    70	    name = _said(parsed, _WORKFLOW)
    71	    label = RunLabel(_said(parsed, _LABEL))
    72	    project, services = registered()
    73	    asyncio.run(
    74	        api.run(
    75	            services,
    76	            project,
    77	            name,
    78	            label,
    79	            argv,
    80	            base_ref=_perhaps(parsed, _BASE_REF),
    81	            points=points,
    82	        )
    83	    )
    84	    print(f"run {str(label)!r} finished")
    85	    return _NOTHING_TO_REPORT
    86	
    87	
    88	def _said(parsed: argparse.Namespace, dest: str) -> str:
    89	    value = getattr(parsed, dest)
    90	    if isinstance(value, str):
    91	        return value
    92	    raise InternalError(
    93	        f"the `{NAME}` parser produced {value!r} for {dest!r}, and every argument this module "
    94	        f"declares is a string it took off the command line. That is AGL's own bug: the parser and "
    95	        f"the reader are in one file and they disagree"
    96	    )

    98	
    99	def _perhaps(parsed: argparse.Namespace, dest: str) -> str | None:
   100	    value = getattr(parsed, dest)
   101	    return None if value is None else _said(parsed, dest)

########## src/agl/cli/exit_codes.py ##########
     1	
     2	from collections.abc import Iterator
     3	
     4	from agl.ports.errors import EXIT_CODES, AglError, InternalError, exit_code_for
     5	
     6	__all__ = ["EXIT_CODES", "exit_code_for", "exit_status", "leaves"]
     7	
     8	
     9	def exit_status(error: Exception) -> int:
    10	    agreed, *disagreeing = {_resolved(leaf) for leaf in leaves(error)}
    11	    return exit_code_for(InternalError) if disagreeing else agreed
    12	
    13	
    14	def leaves(error: Exception) -> Iterator[Exception]:
    15	    if not isinstance(error, ExceptionGroup):
    16	        yield error
    17	        return
    18	    for held in error.exceptions:
    19	        yield from leaves(held)
    20	
    21	
    22	def _resolved(leaf: Exception) -> int:
    23	    if isinstance(leaf, AglError):
    24	        return exit_code_for(leaf)
    25	    return exit_code_for(InternalError)
