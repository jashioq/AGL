    31	def _describe(character: str) -> str:
    32	    name = unicodedata.name(character, "")
    33	    return f"{character!r} ({name})" if name else f"{character!r} (U+{ord(character):04X})"
    34	
    35	
    36	def _collision_key(value: str) -> str:
    37	    return unicodedata.normalize("NFC", value.casefold())
    38	
    39	
    40	def _unusable(value: str) -> str | None:
    41	    if not value:
    42	        return "it is empty"
    43	    if value in {".", ".."}:
    44	        return "it is a path traversal segment, not a name"
    45	
    46	    for index, character in enumerate(value):
    47	        if character in _PATH_SEPARATORS:
    48	            return (
    49	                f"it contains {character!r} at position {index}, and a name is one path "
    50	                f"segment and one git ref component - never a path"
    51	            )
    52	        if unicodedata.category(character)[0] in _INVISIBLE_CATEGORIES:
    53	            return (
    54	                f"it contains {_describe(character)} at position {index}, and a name carries "
    55	                f"no spaces, control characters or other invisible characters"
    56	            )
    57	        if character not in _ALLOWED_CHARACTERS:
    58	            return (
    59	                f"it contains {_describe(character)} at position {index}, and a name may hold "
    60	                f"only letters A-Z a-z, digits, and '.', '_' or '-'"
    61	            )
    62	
    63	    length = len(value.encode("utf-8"))
    64	    if length > _MAX_BYTES:
    65	        return f"it is {length} bytes long, and a path segment may not exceed {_MAX_BYTES}"
    66	
    67	    if value.startswith("-"):
    68	        return "it starts with '-', which any command taking it as an argument reads as a flag"
    69	    if value.endswith("-"):
    70	        return "it ends with '-', which reads as a truncated name and is no use to anyone"
    71	    if ".." in value:
    72	        return "it contains '..', which git ref names may not, and which means a parent directory"
    73	    if value.startswith("."):
    74	        return "it starts with '.', which git ref components may not, and which hides it on disk"
    75	    if value.endswith("."):
    76	        return "it ends with '.', which git ref names may not, and which Windows silently strips"
    77	    if value.lower().endswith(".lock"):
    78	        return "it ends with '.lock', which git reserves for its own lock files"
    79	
    80	    stem = value.split(".", 1)[0]
    81	    if stem.upper() in _RESERVED_DEVICE_NAMES:
    82	        return (
    83	            f"its first component {stem!r} is a reserved device name on Windows, where opening "
    84	            f"the file opens the device instead"
    85	        )
    86	    return None

    98	            workflow_version=_wire_text(data, "workflow_version"),
    99	            label=label,
   100	            base_ref=_wire_text(data, "base_ref"),
   101	            base_sha=_wire_text(data, "base_sha"),
   102	            branch=_wire_text(data, "branch"),
   103	            params=params,
   104	            created_at=created_at,
   105	        )

   108	def _wire_text(data: Mapping[str, object], key: str) -> str:
   109	    value = data[key]
   110	    if not isinstance(value, str):
   111	        raise InternalError(
   112	            f"run.json's {key!r} is a {type(value).__name__}, and a record's {key} is a string"
   113	        )
   114	    return value
   115	
   116	
   117	def _check_sha(value: str) -> None:
   118	    if len(value) not in _SHA_LENGTHS or not _SHA_CHARACTERS.issuperset(value):
   119	        raise InternalError(
   120	            f"base_sha {value!r} is not a resolved commit: expected 40 characters of lowercase "
   121	            f"hexadecimal (sha1) or 64 (sha256) - a full object id, because an abbreviated one "
   122	            f"stops being unique as the repository grows and so pins nothing"
   123	        )
   124

   126	def _normalised(moment: datetime) -> datetime:
   127	    if moment.tzinfo is None or moment.utcoffset() is None:
   128	        raise InternalError(
   129	            f"created_at {moment!r} has no timezone, and a wall-clock reading with no place is "
   130	            f"not a moment - a run record carries an instant, written as UTC"
   131	        )
   132	    utc = moment.astimezone(UTC)
   133	    return utc - timedelta(microseconds=utc.microsecond)

   136	def _checked_params(params: Mapping[str, JsonValue]) -> dict[str, JsonValue]:
   137	    return {
   138	        _checked_key(key): _checked_json(value, f"params.{key}") for key, value in params.items()
   139	    }
   140	
   141	
   142	def _checked_key(key: object) -> str:
   143	    if not isinstance(key, str):
   144	        raise InternalError(
   145	            f"the param key {key!r} is a {type(key).__name__}, and a JSON object is keyed by "
   146	            f"strings - writing this record would silently rename it"
   147	        )
   148	    return _checked_text(key, f"the param key {key!r}")
   149	
   150	
   151	def _checked_text(value: str, where: str) -> str:
   152	    for index, character in enumerate(value):
   153	        if unicodedata.category(character) == _SURROGATE:
   154	            raise InternalError(
   155	                f"{where} holds U+{ord(character):04X} at position {index}, which is a surrogate: "
   156	                f"UTF-8 has no encoding for one at all, so the store refuses the write and this "
   157	                f"refuses it here, where the caller still knows what it handed over"
   158	            )
   159	    return value
   160	
   161	
   162	def _checked_json(value: object, where: str) -> JsonValue:
   163	    if value is None or isinstance(value, bool | int):
   164	        return value
   165	    if isinstance(value, str):
   166	        return _checked_text(value, where)
   167	    if isinstance(value, float):
   168	        if not isfinite(value):
   169	            raise InternalError(
   170	                f"{where} is {value!r}, which JSON has no spelling for: it would be written as a "
   171	                f"bare token no reader accepts, and NaN does not even equal itself"
   172	            )
   173	        return value
   174	    if isinstance(value, Mapping):
   175	        return {
   176	            _checked_key(key): _checked_json(item, f"{where}.{key}") for key, item in value.items()
   177	        }
   178	    if isinstance(value, list | tuple):
   179	        return [_checked_json(item, f"{where}[{index}]") for index, item in enumerate(value)]
   180	    raise InternalError(
   181	        f"{where} is a {type(value).__name__}, which is not a JSON value: a workflow's params are "
   182	        f"stored exactly as given and read back the same, so they have to be writable"
   183	    )
