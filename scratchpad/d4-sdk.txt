   135	def prompt_file(path: str | Path) -> str:
   136	    asked = Path(path)
   137	    where = asked if asked.is_absolute() else _beside_the_caller(sys._getframe(1).f_globals, asked)
   138	    try:
   139	        text = where.read_text(encoding="utf-8")
   140	    except FileNotFoundError as missing:
   141	        raise InputError(
   142	            f"there is no prompt file at {where}, so the role declared with it would have nothing "
   143	            f"to ask its agent. A relative path is resolved against the directory of the module "
   144	            f"that called `prompt_file` - never against the current directory - so this is the "
   145	            f"path {asked!r} names from where it was written"
   146	        ) from missing
   147	    except IsADirectoryError as directory:
   148	        raise InputError(
   149	            f"{where} is a directory, and a role's instructions are one prompt. Name the file "
   150	            f"inside it that this role is asking its agent to read"
   151	        ) from directory
   152	    except UnicodeDecodeError as undecodable:
   153	        raise InputError(
   154	            f"{where} is not UTF-8 text: {undecodable}. A prompt is read as UTF-8 and nothing "
   155	            f"here repairs one - a prompt read with replacement characters in it is a prompt "
   156	            f"nobody wrote, fingerprinted as though somebody had, and handed to a model that "
   157	            f"would answer it anyway"
   158	        ) from undecodable
   159	    except OSError as unreadable:
   160	        raise InputError(
   161	            f"{where} could not be read: {unreadable}. The prompt is read where the role is "
   162	            f"declared, so this is the role's factory failing rather than the step it was for"
   163	        ) from unreadable
   164	    if not text.strip():
   165	        raise InputError(
   166	            f"the prompt file {where} is empty, and a role's instructions are the whole of what "
   167	            f"its agent is asked to do. `Role` refuses an empty prompt too, one line later, and "
   168	            f"`AgentTask` refuses it again at the dispatch"
   169	        )
   170	    return text

   177	def _beside_the_caller(caller: Mapping[str, object], asked: Path) -> Path:
   178	    declared = caller.get("__file__")
   179	    if not isinstance(declared, str):
   180	        raise InputError(
   181	            f"`prompt_file({str(asked)!r})` was called from something with no `__file__` - a REPL, "
   182	            f"an `exec`, or a frozen import - so there is no module directory for a relative path "
   183	            f"to be relative to. Pass an absolute path. AGL will not fall back to the current "
   184	            f"directory: a workflow is read from wherever it was installed, and the directory this "
   185	            f"process happens to have started in is the one place its prompts are certainly not"
   186	        )
   187	    return Path(declared).resolve().parent / asked

    37

    44	    def __post_init__(self) -> None:
    45	        if not self.name:

    47	        if not self.description:
    48	            raise InputError(

    61	    def read(self, value: object) -> P:
    62	        instance, problems = self._read(value)
    63	        if instance is None:
    64	            raise InternalError(
    65	                f"a recorded {self.name} payload does not fit {_describe(self.payload)}: "
    66	                f"{_refusal(self.name, problems)}. AGL wrote this value and AGL is reading it, so "
    67	                f"the ledger and the payload type have come apart - §3.6 expects the fingerprint "
    68	                f"to have discarded this entry, and `sdk/tools.py` records why it may not have"

    73	        problems: list[str] = []

    75	        if not isinstance(built, self.payload):

   102	def _object_schema(
   103	    kind: type[Any], where: str, inside: tuple[type[Any], ...]
   104	) -> dict[str, JsonValue]:
   105	    if kind in inside:
   106	        raise InputError(
   107	            f"{where} is {_describe(kind)}, which is already being derived - a payload dataclass "
   108	            f"that contains itself has no finite schema, and a model has no way to know when to "
   109	            f"stop nesting one"
   110	        )
   111	    hints = _hints(kind)
   112	    nested = (*inside, kind)
   113	    properties: dict[str, JsonValue] = {}
   114	    required: list[JsonValue] = []
   115	    for spec in fields(kind):
   116	        schema = _schema_for(hints.get(spec.name), _at(where, spec.name), nested)
   117	        described = _description(spec)
   118	        if described is not None:
   119	            schema["description"] = described
   120	        properties[spec.name] = schema
   121	        if _required(spec.default, spec.default_factory):
   122	            required.append(spec.name)
   123	    return {
   124	        "type": "object",
   125	        "title": f"{kind.__module__}.{kind.__qualname__}",
   126	        "properties": properties,
   127	        "required": sorted(required, key=str),
   128	        "additionalProperties": False,
   129	    }

   132	def _schema_for(hint: object, where: str, inside: tuple[type[Any], ...]) -> dict[str, JsonValue]:
   133	    optional = _optional(hint)
   134	    if optional is not None:
   135	        return {"anyOf": [_schema_for(optional, where, inside), {"type": "null"}]}
   136	    scalar = _SCALARS.get(hint)
   137	    if scalar is not None:
   138	        return {"type": scalar[0]}
   139	    item = _item(hint)
   140	    if item is not None:
   141	        return {"type": "array", "items": _schema_for(item, f"{where}[]", inside)}
   142	    if isinstance(hint, type) and is_dataclass(hint):
   143	        return _object_schema(hint, where, inside)
   144	    raise InputError(
   145	        f"{where} is a {_describe(hint)}, which a reporting tool cannot carry: {_SUPPORTED}. A "
   146	        f"payload is stored as the step's result and read back as this dataclass, so a field type "
   147	        f"that does not survive JSON unchanged is one the ledger could not return"
   148	    )

   151	def _instance(kind: type[Any], value: object, where: str, problems: list[str]) -> object:
   152	    before = len(problems)
   153	    given = _given(kind, value, where, problems)
   154	    if len(problems) != before:
   155	        return None
   156	    factory: Callable[..., object] = kind
   157	    try:
   158	        return factory(**given)
   159	    except Exception as raised:
   160	        problems.append(f"`{where}` is not a valid {_describe(kind)}: {raised}")
   161	        return None
   162	
   163	
   164	def _given(kind: type[Any], value: object, where: str, problems: list[str]) -> dict[str, object]:
   165	    if not isinstance(value, Mapping):
   166	        problems.append(f"`{where}` should be an object and {_shown(value)} arrived")
   167	        return {}
   168	    hints = _hints(kind)
   169	    declared = {spec.name: spec for spec in fields(kind)}
   170	    for key in value:
   171	        if key not in declared:
   172	            problems.append(
   173	                f"`{_at(where, key)}` is not one of this payload's fields. They are: "
   174	                f"{', '.join(declared) or '(none)'}"
   175	            )
   176	    given: dict[str, object] = {}
   177	    for name, spec in declared.items():
   178	        hint, at = hints.get(name), _at(where, name)
   179	        if name in value:
   180	            given[name] = _converted(hint, value[name], at, problems)
   181	        elif _required(spec.default, spec.default_factory):
   182	            problems.append(f"`{at}` is missing, and it is required: {_wording(hint)}")
   183	    return given
   184

   186	def _converted(hint: object, value: object, where: str, problems: list[str]) -> object:
   187	    optional = _optional(hint)
   188	    if optional is not None:
   189	        return None if value is None else _converted(optional, value, where, problems)
   190	    if hint in _SCALARS:
   191	        return _scalar(hint, value, where, problems)
   192	    item = _item(hint)
   193	    if item is not None:
   194	        if not isinstance(value, list):
   195	            _wrong(hint, value, where, problems)
   196	            return None
   197	        built = [
   198	            _converted(item, element, f"{where}[{index}]", problems)
   199	            for index, element in enumerate(value)
   200	        ]
   201	        return tuple(built) if get_origin(hint) is tuple else built
   202	    if isinstance(hint, type) and is_dataclass(hint):
   203	        return _instance(hint, value, where, problems)
   204	    _wrong(hint, value, where, problems)
   205	    return None
   206

   208	def _scalar(hint: object, value: object, where: str, problems: list[str]) -> object:
   209	    if hint is str and isinstance(value, str):
   210	        return value
   211	    if hint is bool and isinstance(value, bool):
   212	        return value
   213	    if hint is int and isinstance(value, int) and not isinstance(value, bool):
   214	        return value
   215	    if (
   216	        hint is float
   217	        and isinstance(value, int | float)
   218	        and not isinstance(value, bool)
   219	        and isfinite(value)
   220	    ):
   221	        return float(value)
   222	    _wrong(hint, value, where, problems)
   223	    return None
   224

   226	def _wrong(hint: object, value: object, where: str, problems: list[str]) -> None:
   227	    problems.append(f"`{where}` should be {_wording(hint)} and {_shown(value)} arrived")
   228

   230	def _wording(hint: object) -> str:
   231	    optional = _optional(hint)
   232	    if optional is not None:
   233	        return f"{_wording(optional)} or null"
   234	    scalar = _SCALARS.get(hint)
   235	    if scalar is not None:
   236	        return scalar[1]
   237	    item = _item(hint)
   238	    if item is not None:
   239	        return f"an array of {_wording(item)}"
   240	    if isinstance(hint, type) and is_dataclass(hint):
   241	        return f"an object with the fields of {hint.__qualname__}"
   242	    return f"a {_describe(hint)}, which is not a payload type at all"
   243	
   244	
   245	def _optional(hint: object) -> object | None:
   246	    if get_origin(hint) is not UnionType:
   247	        return None
   248	    args: tuple[object, ...] = get_args(hint)
   249	    real = [arg for arg in args if arg is not type(None)]
   250	    return real[0] if len(args) == 2 and len(real) == 1 else None
   251	
   252	
   253	def _item(hint: object) -> object | None:
   254	    origin = get_origin(hint)
   255	    args: tuple[object, ...] = get_args(hint)
   256	    if origin is list and len(args) == 1:
   257	        return args[0]
   258	    if origin is tuple and len(args) == 2 and args[1] is Ellipsis:
   259	        return args[0]
   260	    return None

   263	def _check_payload(payload: object, name: str) -> None:
   264	    if isinstance(payload, type) and is_dataclass(payload):
   265	        return
   266	    raise InputError(

   268	        f"dataclass (§3.3) - the class itself, never an instance of it. Its fields are what the "
   269	        f"schema is derived from and what the agent is asked to fill in"
   270	    )

   273	def _hints(kind: type[Any]) -> Mapping[str, object]:
   274	    try:
   275	        return get_type_hints(kind)
   276	    except (NameError, TypeError) as error:
   277	        raise InputError(
   278	            f"{_describe(kind)} has an annotation that cannot be resolved: {error}. Its fields are "
   279	            f"read for their types, so each has to name something importable where it is"

   283	def _refusal(name: str, problems: tuple[str, ...]) -> str:
   284	    listed = "\n".join(f"  - {problem}" for problem in problems)
   285	    return (
   286	        f"{name} was not given a payload it can accept, and nothing has been recorded. Fix all of "
   287	        f"these and call it again:\n{listed}"
   288	    )

   291	def _required(default: object, factory: object) -> bool:

   169	def _declared(fn: Callable[..., object]) -> object:
   170	    parameters, hints = _hints(fn)
   171	    if not parameters:
   172	        raise InputError(

   174	            f"function taking a `Run` (§3.3) - which is also where it declares its own parameters, "
   175	            f"now that `@workflow` takes only `version=`. Write `async def {fn.__qualname__}(run: "
   176	            f"Run[YourParams])`, or `run: Run` for a workflow that never reads `run.params`"
   177	        )
   178	    first = parameters[0]
   179	    if first not in hints:

   182	            f"and that annotation is the one place a workflow declares its parameters (§3.3). An "
   183	            f"unannotated parameter is **not** read as a bare `Run`: the two say different things "
   184	            f"and only one of them was written down. Write `{first}: Run[YourParams]`, or "
   185	            f"`{first}: Run` for a workflow that never reads `run.params`"
   186	        )
   187	    annotation = hints[first]
   188	    origin = get_origin(annotation)
   189	    subject = annotation if origin is None else origin
   190	    if subject is Run:
   191	        arguments = get_args(annotation)
   192	        return arguments[0] if arguments else _NoParams
   193	    raise InputError(_not_a_run(fn, first, annotation, subject))

   201	            f"class of its own, so the annotation would be describing an object this run cannot "
   202	            f"produce - and the params are read from it precisely because it and the object agree. "
   203	            f"§3.3 gives two spellings and they are the whole list: `Run[YourParams]`, and a bare "
   204	            f"`Run` for a workflow that never reads `run.params`"
   205	        )

    40	def arg(*flags: str, default: Any = MISSING, help: str = "") -> Any:
    41	    if not flags:
    42	        raise InputError(
    43	            "a parameter is declared with at least one flag - `arg('-r', '--request')`. A field "

    61	def parser_for(params: type[object], *, prog: str | None = None) -> RefusingParser:
    62	    if not is_dataclass(params):
    63	        raise InputError(f"{_describe(params)} is not a dataclass of `arg()` fields (§3.3)")
    64	    parser = RefusingParser(prog=prog, add_help=False, allow_abbrev=False)
    65	    hints = _hints(params)
    66	    claimed: dict[str, str] = {}
    67	    for spec in fields(params):
    68	        where = f"{_describe(params)}.{spec.name}"
    69	        declared = spec.metadata.get(_METADATA_KEY)
    70	        if not isinstance(declared, _Declared):
    71	            raise InputError(
    72	                f"{where} is not declared with `arg()`, and every field of a params dataclass is a "
    73	                f"flag: a field without one could never be given a value on the command line"
    74	            )
    75	        for flag in declared.flags:
    76	            if claimed.setdefault(flag, spec.name) != spec.name:
    77	                raise InputError(
    78	                    f"{where} and its sibling {claimed[flag]!r} both declare {flag!r}, so one of "
    79	                    f"the two could never receive a value"
    80	                )
    81	        parser.add_argument(
    82	            *declared.flags, dest=spec.name, required=spec.default is MISSING,
    83	            default=argparse.SUPPRESS, help=declared.help or None,
    84	            **_consumes(where, hints.get(spec.name), spec.default),
    85	        )

   195	def _unusable_flag(flag: str) -> str | None:
   196	    if not flag.startswith("-"):
   197	        return "it is a positional, and `agl run <workflow>` already occupies that position (§3.3)"
   198	    name = flag.removeprefix("--") if flag.startswith("--") else flag[1:]

   200	        return "it is dashes and nothing else"
   201	    if not flag.startswith("--") and len(name) != 1:

   203	    if name[0] not in ascii_letters:
   204	        return "a flag's name starts with an ASCII letter"
   205	    for index, character in enumerate(name):
   206	        if character not in _FLAG_CHARACTERS:
   207	            return (
   208	                f"it holds {character!r} at position {index}, and a flag may hold only letters "
   209	                f"A-Z a-z, digits, '-' and '_'"
   210	            )
   211	    return None
