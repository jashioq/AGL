     5	from agl.sdk import Run, arg, workflow
     6	from agl.workflows.split import views
     7	from agl.workflows.split.chunks import Chunk
     8	from agl.workflows.split.roles import implementer, planner
     9	
    10	__all__ = ["SplitParams", "split"]
    11	
    12	
    13	@dataclass(frozen=True, slots=True)
    14	class SplitParams:
    15	
    16	    request: str = arg("-r", "--request", help="the job to divide up, in your own words")
    17	
    18	    chunks: int = arg("-c", "--chunks", default=3, help="the most chunks to divide it into")
    19	
    20	
    21	@workflow(version="1.1")
    22	async def split(run: Run[SplitParams]) -> None:
    23	    plan = await run.step(planner(), request=run.params.request, chunks=run.params.chunks)
    24	    children = {chunk.id: run.worktree(chunk.id) for chunk in plan.items}
    25	    await run.terminal.show(views.board, chunks=plan.items, runs=children)
    26	    async with TaskGroup() as group:
    27	        for chunk in plan.items:
    28	            group.create_task(_implement(children[chunk.id], chunk))
    29	
    30	
    31	async def _implement(w: Run[SplitParams], chunk: Chunk) -> None:
    32	    await w.step(implementer(), chunk=chunk, commit=f"implement {chunk.id}")
    33	    outcome = await w.integrate()
    34	    while outcome.conflicted:
    35	        if await w.terminal.show(
    36	            views.conflict, conflict=outcome.conflict, build=outcome.verdict, priority=10
    37	        ):
    38	            await outcome.retry()
    39	            continue
    40	        await outcome.abort()
    41	        break

     1	
     2	from dataclasses import dataclass
     3	from typing import Final
     4	
     5	from agl.sdk import Namespace, describe, reporting_tool
     6	
     7	__all__ = ["Chunk", "Chunks", "report_chunks"]
     8	
     9	
    10	@dataclass(frozen=True, slots=True)
    11	class Chunk:
    12	
    13	    id: str = describe(
    14	        "This chunk's name. It becomes a git branch and a directory, so it may hold only letters "
    15	        "A-Z a-z, digits, '.', '_' and '-' - no spaces, no slashes, no leading or trailing '.' or "
    16	        "'-' - it may not be '_base', and no two chunks may share one, compared without regard to "
    17	        "case. Keep it short and descriptive: a person reads these as branch names afterwards."
    18	    )
    19	
    20	    work: str = describe(
    21	        "The whole assignment for this chunk, written for an agent that will never see the rest of "
    22	        "this plan and has no memory of the reasoning behind it."
    23	    )
    24	
    25	    files: tuple[str, ...] = describe(
    26	        "The paths this chunk should touch, relative to the repository root, including ones it "
    27	        "will create. Two chunks naming one file will collide when their work is merged back."
    28	    )
    29	
    30	    def __post_init__(self) -> None:
    31	        Namespace(self.id)
    32	
    33	
    34	@dataclass(frozen=True, slots=True)
    35	class Chunks:
    36	
    37	    items: tuple[Chunk, ...] = describe(
    38	        "Every chunk this job divides into. Report at least one: a job too small to divide is a "
    39	        "plan with one chunk in it."
    40	    )
    41	
    42	    def __post_init__(self) -> None:
    43	        if not self.items:
    44	            raise ValueError(
    45	                "a plan with no chunks in it is not a plan: every chunk is a worktree, so this "
    46	                "would be a run that opened nothing, ran nothing and reported success over a job "
    47	                "nobody did. There is always at least one chunk - a job too small to divide is a "
    48	                "plan with one chunk in it, and it runs in one worktree"
    49	            )
    50	        taken: dict[str, str] = {}
    51	        for chunk in self.items:
    52	            key = Namespace(chunk.id).collision_key
    53	            first = taken.get(key)
    54	            if first is not None:
    55	                raise ValueError(
    56	                    f"two chunks are named {first!r} and {chunk.id!r}, which are two names to git "
    57	                    f"and one directory on a case-insensitive filesystem, so the second worktree "
    58	                    f"would be the first one's checkout. Every chunk needs a name no other chunk "
    59	                    f"shares, compared without regard to case"
    60	                )
    61	            taken[key] = chunk.id
    62	
    63	
    64	report_chunks: Final = reporting_tool(
    65	    "report_chunks",
    66	    "Report the chunks this job divides into, and end the planning step. Call this exactly once, "
    67	    "when the plan is complete: it is the only way to record a result, and a planning step that "
    68	    "ends without calling it has produced nothing and will be run again.",
    69	    Chunks,
    70	)

    10	def board(chunks: Sequence[Chunk], runs: Mapping[str, Run]) -> Screen:
    11	    return Screen(Rows([Row(chunk.id, _line(runs.get(chunk.id))) for chunk in chunks]))
    12	
    13	
    14	def _line(run: Run | None) -> str:
    15	    return "" if run is None else run.activity or ""
########## src/agl/workflows/split/views/conflict.py ##########
     1	
     2	from typing import Final
     3	
     4	from agl.sdk import Choice, Conflict, Row, Rows, Screen, VerifierOutcome
     5	
     6	__all__ = ["ABORT", "RETRY", "conflict"]
     7	
     8	RETRY: Final = "Try landing it again"
     9	
    10	ABORT: Final = "Give up on this chunk - its branch and worktree stay"
    11	
    12	_TAIL: Final = 2_000
    13	
    14	
    15	def conflict(conflict: Conflict, build: VerifierOutcome | None) -> Screen[bool]:
    16	    rows = [Row(conflict.summary), *(Row(path) for path in conflict.paths)]
    17	    if build is not None:
    18	        rows.append(Row(f"The build exited {build.status}, and its output ended like this:"))
    19	        rows.extend(Row(line) for line in _tail(build.output))
    20	    return Screen(Rows(rows), [Choice(RETRY, value=True), Choice(ABORT, value=False)])
    21	
    22	
    23	def _tail(output: str) -> list[str]:
    24	    kept = output[-_TAIL:]
    25	    lines = kept.splitlines()
    26	    if lines and len(kept) < len(output) and output[-_TAIL - 1] not in "\r\n":
    27	        del lines[0]
    28	    return lines
