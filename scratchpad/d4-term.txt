    14	type View[T] = Callable[..., Screen[T]]

    17	@dataclass(frozen=True, slots=True, eq=False)
    18	class Registration:
    19	
    20	    view: View[object]
    21	
    22	    params: Mapping[str, object]
    23	
    24	    def screen(self) -> Screen[object]:
    25	        return self.view(**self.params)
    26	
    27	
    28	@dataclass(frozen=True, slots=True, eq=False)
    29	class Entry(Registration):
    30	
    31	    priority: int
    32	
    33	    answered: asyncio.Future[object]
    34	
    35	
    36	class Screens:
    37	
    38	    __slots__ = ("_queues", "_slot")
    39	
    40	    def __init__(self) -> None:
    41	        self._slot: Registration | None = None
    42	        self._queues: dict[int, list[Entry]] = {}
    43	
    44	    def hold(self, view: View[object], /, **params: object) -> None:
    45	        self._slot = Registration(view, params)
    46	
    47	    def queue[T](self, view: View[T], /, *, priority: int = 0, **params: object) -> Awaitable[T]:
    48	        answered: asyncio.Future[object] = asyncio.get_running_loop().create_future()
    49	        entry = Entry(view, params, priority, answered)
    50	        self._queues.setdefault(priority, []).append(entry)
    51	        answered.add_done_callback(lambda _: self._retire(entry))
    52	        return cast("Awaitable[T]", answered)
    53	
    54	    @property
    55	    def slot(self) -> Registration | None:
    56	        return self._slot
    57	
    58	    @property
    59	    def current(self) -> Entry | None:
    60	        busy = (priority for priority, entries in self._queues.items() if entries)
    61	        busiest = max(busy, default=None)
    62	        return None if busiest is None else self._queues[busiest][0]
    63	
    64	    @property
    65	    def displayed(self) -> Registration | None:
    66	        current = self.current
    67	        return self._slot if current is None else current
    68	
    69	    @property
    70	    def pending(self) -> Mapping[int, int]:
    71	        counts = {priority: len(entries) for priority, entries in self._queues.items()}
    72	        current = self.current
    73	        if current is not None:
    74	            counts[current.priority] -= 1
    75	        return MappingProxyType(counts)
    76	
    77	    def answer(self, response: int, typed: str = "") -> None:
    78	        entry = self.current
    79	        if entry is None:
    80	            raise InternalError(
    81	                f"answered with response {response} while no question was displayed. Only the "
    82	                f"displayed screen can be answered and the terminal knows which one that is, "
    83	                f"so an answer arriving with nothing to answer is AGL's own ordering rather "
    84	                f"than anything a workflow author did"
    85	            )
    86	        offered = entry.screen().responses
    87	        if not 0 <= response < len(offered):
    88	            raise InternalError(
    89	                f"answered with response {response} on a screen offering {len(offered)}. A "
    90	                f"response is named by its position in the order the view offered them, and the "
    91	                f"only thing that names one is the terminal drawing that same screen"
    92	            )
    93	        self._retire(entry)
    94	        if entry.answered.done():
    95	            return
    96	        try:
    97	            given = _given(offered[response], typed)
    98	        except Exception as failed:
    99	            entry.answered.set_exception(failed)
   100	        else:
   101	            entry.answered.set_result(given)
   102	
   103	    def close(self) -> None:
   104	        for entries in self._queues.values():
   105	            for entry in tuple(entries):
   106	                if not entry.answered.done():
   107	                    entry.answered.set_exception(
   108	                        UpstreamUnavailable(
   109	                            "the terminal shut down with this question still waiting for an "
   110	                            "answer, so nobody can ever give one"
   111	                        )
   112	                    )
   113	            entries.clear()
   114	        self._slot = None
   115	
   116	    def _retire(self, entry: Entry) -> None:
   117	        entries = self._queues.get(entry.priority)
   118	        if entries is not None and entry in entries:
   119	            entries.remove(entry)
   120	
   121	
   122	def _given(chosen: Response[object], typed: str) -> object:
   123	    match chosen:
   124	        case Choice():
   125	            return chosen.value
   126	        case TextInput():
   127	            return chosen.maps(typed)

   126	    async def __aenter__(self) -> Self:
   127	        if self._open:
   128	            raise InternalError(
   129	                "this terminal is already inside its context. Entering twice would start a second "
   130	                "redraw loop and a second reader over the same queues, and the first pair would "
   131	                "keep drawing after the terminal was handed back - the framework opens a terminal "
   132	                "once, so arriving here twice is AGL's own ordering bug"
   133	            )
   134	        self._display.start()
   135	        self._reads = ThreadPoolExecutor(max_workers=1, thread_name_prefix="agl-terminal-keys")
   136	        self._open = True
   137	        self._tasks = (
   138	            asyncio.create_task(self._redrawing()),
   139	            asyncio.create_task(self._answering()),
   140	        )
   141	        return self
   142	
   143	    async def __aexit__(
   144	        self,
   145	        exc_type: type[BaseException] | None,
   146	        exc: BaseException | None,
   147	        tb: TracebackType | None,
   148	    ) -> None:
   149	        self._open = False
   150	        try:
   151	            self._keys.stop()
   152	            for task in self._tasks:
   153	                task.cancel()
   154	            await asyncio.gather(*self._tasks, return_exceptions=True)
   155	            self._tasks = ()
   156	            if self._reads is not None:
   157	                self._reads.shutdown(wait=False, cancel_futures=True)
   158	                self._reads = None
   159	            self._screens.close()
   160	        finally:
   161	            self._display.stop()
   162	            self._written = None
   163	
   164	    async def _redrawing(self) -> None:
   165	        while True:
   166	            self._draw()
   167	            await asyncio.sleep(self._period)
   168	
   169	    def _draw(self) -> None:
   170	        if self._taken:
   171	            return
   172	        registration = self._screens.displayed
   173	        screen = None if registration is None else self._screen_of(registration)
   174	        if screen == self._written:
   175	            return
   176	        self._display.write(None if screen is None else frame(screen))
   177	        self._written = screen
   178	
   179	    def _screen_of(self, registration: Registration) -> Screen[object]:
   180	        try:
   181	            return registration.screen()
   182	        except Exception as failed:
   183	            named = getattr(registration.view, "__name__", repr(registration.view))
   184	            return Screen(Text(f"the view {named} could not be drawn: {failed!r}"))
   185	
   186	    async def _answering(self) -> None:
   187	        while True:
   188	            if self._screens.current is None:
   189	                await asyncio.sleep(self._period)
   190	                continue
   191	            line = await self._read()
   192	            if line is None:
   193	                return
   194	            await self._answer(line)
   195	
   196	    async def _answer(self, line: str) -> None:
   197	        entry = self._screens.current
   198	        if entry is None:
   199	            return
   200	        offered = self._screen_of(entry).responses
   201	        position = _position(line, len(offered))
   202	        if position is None:
   203	            return
   204	        typed = ""
   205	        picked = offered[position]
   206	        if isinstance(picked, TextInput):
   207	            read = await self._typed(picked)
   208	            if read is None:
   209	                return
   210	            typed = read
   211	        if self._screens.current is entry:
   212	            self._screens.answer(position, typed)
   213	
   214	    async def _typed(self, field: TextInput[object]) -> str | None:
   215	        self._taken = True
   216	        self._display.release()
   217	        try:
   218	            self._console.print(RichText(f"{field.label}:"))
   219	            return await self._read()
   220	        finally:
   221	            self._written = None
   222	            self._display.resume()
   223	            self._taken = False
   224	
   225	    async def _read(self) -> str | None:
   226	        reads = self._reads
   227	        if reads is None:
   228	            return None
   229	        try:
   230	            return await asyncio.get_running_loop().run_in_executor(reads, self._keys.read)
   231	        except RuntimeError:
   232	            return None
   233	
   234	
   235	def _position(line: str, offered: int) -> int | None:
   236	    picked = line.strip()
   237	    if not picked.isdigit():
   238	        return None
   239	    position = int(picked) - 1
   240	    return position if 0 <= position < offered else None

    19	    body = drawn(screen.body)
    20	    if not screen.responses:
    21	        return body
    22	    return Group(body, RichText(""), _offered(screen.responses), RichText(""), _prompt(screen))
    23	
    24	
    25	def drawn(component: Component) -> RenderableType:
    26	    match component:
    27	        case Text():
    28	            return RichText(component.value)
    29	        case Row():
    30	            return _grid([component])
    31	        case Rows():
    32	            return _grid(component.rows)
    33	
    34	
    35	def _grid(rows: Sequence[Row]) -> Table:
    36	    grid = Table.grid(padding=_GAP)
    37	    for row in rows:
    38	        grid.add_row(*(RichText(cell.value) for cell in row.cells))
    39	    return grid
    40	
    41	
    42	def _offered(responses: tuple[Response[object], ...]) -> Table:
    43	    listing = Table.grid(padding=_GAP)
    44	    for position, response in enumerate(responses, start=1):
    45	        listing.add_row(RichText(f"{position}."), RichText(_label(response)))
    46	    return listing
    47	
    48	
    49	def _label(response: Response[object]) -> str:
    50	    match response:
    51	        case Choice():
    52	            return response.label
    53	        case TextInput():
    54	            return f"{response.label} ({_TYPE_A_LINE})"
    55	
    56	
    57	def _prompt(screen: Screen[object]) -> RichText:
    58	    return RichText(f"[1-{len(screen.responses)}] and Enter:")
